T0 = time.perf_counter()


def say(tag, obj):
    print(f"[{time.perf_counter() - T0:7.2f}s] {tag}: {json.dumps(obj, sort_keys=True, default=str)}")


k = kernel_bridge.kernel()

# lr
tr = dev[:4096]
X = pipeline.normalized_vectors(tr)
y = labels_of(tr)
fit = lr.fit(lr.columns_from_vectors(X), y, 1e-3)
say("lr.fit l2=1e-3", {kk: fit[kk] for kk in ("iterations", "converged", "stop_reason", "gradient_max_abs", "step_halvings", "solvers")})
say("lr.recovery", truth.recovery_report(fit["intercept"], fit["weights"]))

# calibration
raw_art, _ = pipeline.build_candidate(train_rows=tr, calibration_rows=dev[4096:6144], conformal_rows=dev[6144:8192],
                                      threshold_rows=dev[8192:10240], l2=1e-3, calibrator="none",
                                      split_names={r: f"dev-{r}" for r in pipeline.SPLIT_ROLES})
sc = k.ArtifactScorer(raw_art)
cal_raw = [sc.evaluate(v)["raw_score"] for v in pipeline.normalized_vectors(dev[4096:6144])]
cal_y = labels_of(dev[4096:6144])
p, pd = calibration.fit_platt(cal_raw, cal_y)
say("platt", {"params": p["params"], "diag": pd})
iso, idg = calibration.fit_isotonic(cal_raw, cal_y)
vals = iso["params"]["values"]
say("isotonic", {"blocks": idg["blocks"], "tie_blocks": idg["tie_blocks"], "strictly_increasing": all(b > a for a, b in zip(vals, vals[1:]))})

# pipeline with each calibrator
arts = {}
for kind in calibration.KINDS:
    art, diag = pipeline.build_candidate(train_rows=tr, calibration_rows=dev[4096:6144], conformal_rows=dev[6144:8192],
                                         threshold_rows=dev[8192:10240], l2=1e-3, calibrator=kind,
                                         split_names={r: f"dev-{r}" for r in pipeline.SPLIT_ROLES})
    arts[kind] = art
    say(f"build_candidate[{kind}]", {"threshold": art["decision_threshold"], "q_hat": art["conformal"]["q_hat"],
                                     "n_cal": art["conformal"]["n_calibration"], "conformal_rank": diag["conformal"]["rank"]})

# metrics on held-out DEV rows
ev = dev[10240:12288]
ey = labels_of(ev)
for kind, art in arts.items():
    s = k.ArtifactScorer(art)
    res = [s.evaluate(v) for v in pipeline.normalized_vectors(ev)]
    scores = [r["calibrated_score"] for r in res]
    dec = [1 if r["threshold_decision"] else 0 for r in res]
    val = metrics.baseline_validity(ey, dec, receipt_verified=True, probes_refused=1, probes_total=1)
    rep = metrics.binary_report(ey, dec, val)
    sr = metrics.score_report(ey, scores)
    cov = metrics.coverage(ey, [r["prediction_set"] for r in res])
    sel = metrics.selective(ey, [r["advisory"] for r in res], [r["abstain_reason"] for r in res])
    say(f"metrics[{kind}]", {"ba": rep["balanced_accuracy"], "per_class": rep["per_class_rates"], "brier": sr["brier"],
                             "ece": sr["ece"], "auc": sr["roc_auc"], "coverage": cov, "abstain": sel["abstention_rate"]})
const = metrics.binary_report(ey, [1] * len(ey), metrics.baseline_validity(ey, [1] * len(ey), receipt_verified=True, probes_refused=1, probes_total=1))
say("metrics[constant_alert]", {"ba": const["balanced_accuracy"], "per_class": const["per_class_rates"], "failed": const["baseline_validity"]["failed_preconditions"]})
say("metrics[v1 counts 40/57/132/11]", metrics.rates_from_counts(40, 57, 132, 11))

# freeze (test-only override: tree is dirty during stage 1) + kernel load + advise
with tempfile.TemporaryDirectory() as tmp:
    try:
        freeze.freeze(arts["platt"], Path(tmp) / "real", selection={"procedure": "smoke", "selection_split": "dev", "trials": 0, "selected_trial": None, "stop_reason": None})
        say("freeze(real git probe)", "WROTE (tree clean)")
    except freeze.FreezeRefused as exc:
        say("freeze(real git probe)", f"REFUSED: {str(exc)[:160]}")
    out = freeze.freeze(arts["platt"], Path(tmp) / "a", selection={"procedure": "smoke", "selection_split": "dev", "trials": 0, "selected_trial": None, "stop_reason": None},
                        source_state_override={"commit": "0" * 40, "tree_clean": True})
    kern = k.OpsHealthKernel(out["model_path"], out["receipt_path"])
    adv = kern.advise({"features": dict(gates.VALID_FEATURES)})
    say("kernel.advise(valid)", {kk: adv[kk] for kk in ("advisory", "abstain_reason", "operator_attention_score", "authority")})
    fc, rep = gates.g_failclosed(kern.advise)
    say("g_failclosed", {"fraction": fc, "total": rep["total"], "escaped": rep["escaped"]})
    gi = gates.g_integrity(out["model_path"], out["receipt_path"], k)
    say("g_integrity", gi)
    mono, mrep = gates.g_monotone(lambda f: kern.advise({"features": f})["operator_attention_score"])
    say("g_monotone", {"value": mono, **mrep})
    say("g_truth_sign", gates.g_truth_sign(arts["platt"]["weights"]))

# bootstrap (tiny)
res = [k.ArtifactScorer(arts["platt"]).evaluate(v) for v in pipeline.normalized_vectors(ev)]
bs = bootstrap.BinaryStats(ey, [1 if r["threshold_decision"] else 0 for r in res])

vals = bootstrap.paired_bootstrap(len(ey), {"ba": bs.balanced_accuracy, "ece": ss.ece, "auc": ss.auroc}, seed=20260927, resamples=200)
say("bootstrap(200)", {kk: bootstrap.interval(v) for kk, v in vals.items()})
ones = [1] * len(ey)
say("bootstrap unit-weight equality", {"auc": ss.auroc(ones) == metrics.roc_auc(ey, [r["calibrated_score"] for r in res]),
